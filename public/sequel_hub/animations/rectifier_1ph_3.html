import gseim_calc as calc
import numpy as np

s_C = "1500e-6" # to be changed by user
s_I0 = "31.416"

l = [
  ('$C', s_C),
  ('$I0', s_I0),
]
calc.replace_strings_1("rectifier_1ph_3_orig.in", "rectifier_1ph_3.in", l)
print('rectifier_1ph_3.in is ready for execution')

import numpy as np
import matplotlib.pyplot as plt 
import gseim_calc as calc
from setsize import set_size

f_hz = 50.0
T = 1.0/f_hz

slv = calc.slv("rectifier_1ph_3.in")

i_slv = 0
i_out = 0
filename = slv.l_filename_all[i_slv][i_out]
print('filename:', filename)
u = np.loadtxt(filename)
t = u[:, 0]

ID  = slv.get_array_double(i_slv,i_out,"ID",u)
v_s = slv.get_array_double(i_slv,i_out,"v_s",u)
v_c = slv.get_array_double(i_slv,i_out,"v_c",u)

color1='blue'
color2='green'
color3='red'

fig, ax = plt.subplots(3, sharex=False)
plt.subplots_adjust(wspace=0, hspace=0.0)

set_size(5.5, 6, ax[0])

for i in range(3):
    ax[i].set_xlim(left=0.0, right=2.0*T*1e3)
    ax[i].grid(color='#CCCCCC', linestyle='solid', linewidth=0.5)

ax[0].set_ylabel(r'$v_s$',   fontsize=12)
ax[1].set_ylabel(r'$I_{D}$', fontsize=12)
ax[2].set_ylabel(r'$v_c$',   fontsize=12)

ax[0].tick_params(labelbottom=False)
ax[1].tick_params(labelbottom=False)

ax[0].plot(t*1e3, v_s, color=color1, linewidth=1.0, label="$v_s$")
ax[1].plot(t*1e3, ID , color=color2, linewidth=1.0, label="$I_{D}$")
ax[2].plot(t*1e3, v_c, color=color3, linewidth=1.0, label="$v_c$")

ax[2].set_xlabel('time (msec)', fontsize=11)

#plt.tight_layout()
plt.show()
