Parameter Value

Current at $P_{max}$ (A) 9.59
$V_{OC}$ at STC (V) 40.9
$I_{SC}$ at STC (A) 10.15
Temperature coefficient for $I_{SC}$ ($\%/^oC$) 0.058
Temperature coefficient for $V_{OC}$ ($\%/^oC$) -0.32
Number of cells in series ($N_C$) 60

In a PV plant, ten such panels, along with bypass diodes (assumed to be ideal), are connected in series as shown in the figure. The irradiation for panels PV1-PV9 is $1000\,$W/m$^2$ and that for PV10 is $500\,$W/m$^2$ because of shading.

import gseim_calc as calc

s_g_rad_1 = '1000'
s_g_rad_2 = '500'
s_t_C = '25'
s_Nc = '60'
s_Voc_ref = '40.9'
s_Isc_ref = '10.15'
s_Vm_ref = '33.4'
s_Im_ref = '9.59'
s_coef_Isc = '0.058'
s_coef_Voc = '-0.32'

l = [
  ('$g_rad_1', s_g_rad_1),
  ('$g_rad_2', s_g_rad_2),
  ('$t_C', s_t_C),
  ('$Nc', s_Nc),
  ('$Voc_ref', s_Voc_ref),
  ('$Isc_ref', s_Isc_ref),
  ('$Vm_ref', s_Vm_ref),
  ('$Im_ref', s_Im_ref),
  ('$coef_Isc', s_coef_Isc),
  ('$coef_Voc', s_coef_Voc),
]
calc.replace_strings_1("solar_6_orig.in", "solar_6.in", l)
print('solar_6.in is ready for execution')

import numpy as np
import matplotlib.pyplot as plt 
import gseim_calc as calc
from setsize import set_size

slv = calc.slv("solar_6.in")

i_slv = 0
i_out = 0
filename = slv.l_filename_all[i_slv][i_out]
print('filename:', filename)
u = np.loadtxt(filename)
t = u[:, 0]

i1   = slv.get_array_double(i_slv,i_out,"i1",u)
i10  = slv.get_array_double(i_slv,i_out,"i10",u)
ID1  = slv.get_array_double(i_slv,i_out,"ID1",u)
ID10 = slv.get_array_double(i_slv,i_out,"ID10",u)
p1   = slv.get_array_double(i_slv,i_out,"p1",u)
p10  = slv.get_array_double(i_slv,i_out,"p10",u)
v_p  = slv.get_array_double(i_slv,i_out,"v_p",u)
P_R1 = slv.get_array_double(i_slv,i_out,"P_R1",u)

v_p = v_p

p_max = (P_R1).max()

print('Maximum power:', "%6.1f"%p_max, "W")

color1 = 'green'
color2 = 'crimson'
color3 = 'blue'
color4 = 'royalblue'
color5 = 'goldenrod'
color6 = 'red'

fig, ax = plt.subplots(4, sharex=False, gridspec_kw={'height_ratios': [1.5, 1.5, 2, 2]})
plt.subplots_adjust(wspace=0, hspace=0.0)

set_size(5.5, 9, ax[0])

ax[0].plot(v_p, i1  , color=color1, linewidth=1.0, label="$I_{PV1}$")
ax[0].plot(v_p, i10 , color=color5, linewidth=1.0, label="$I_{PV10}$")
ax[1].plot(v_p, ID1 , color=color1, linewidth=1.0, label="$I_{D1}$")
ax[1].plot(v_p, ID10, color=color5, linewidth=1.0, label="$I_{D10}$")
ax[2].plot(v_p, p1  , color=color1, linewidth=1.0, label="$P_{PV1}$")
ax[2].plot(v_p, p10 , color=color5, linewidth=1.0, label="$P_{PV10}$")
ax[3].plot(v_p, P_R1, color=color2, linewidth=1.0, label="$P_{total}$")

ax[0].set_ylabel(r'$I_{module}$ (A)', fontsize=12)
ax[1].set_ylabel(r'$I_{diode}$ (A)' , fontsize=12)
ax[2].set_ylabel(r'$P_{module}$ (W)', fontsize=12)
ax[3].set_ylabel(r'$P_{total}$ (W)' , fontsize=12)

for i in range(3):
    ax[i].tick_params(labelbottom=False)

for i in range(4):
    ax[i].grid(color='#CCCCCC', linestyle='solid', linewidth=0.5)
    ax[i].set_xlim(left=0.0)

ax[3].set_xlabel('Total voltage (V)', fontsize=12)
