Flyback converter

The flyback converter shown in the figure has an input voltage of $20\,$V and an output voltage of $40\,$V. The switching frequency of the converter is $50\,$kHz. The flyback transformer is represented using an ideal transformer with a magnetizing inductance $L_m = 100\,\mu$H. The turns ratio of the transformer is 1:4, the load resistance is $10\,\Omega$, and the output capacitance is $100\,\mu$F.
  1. What is the duty cycle of the converter?
  2. Plot the current through $L_m$, and determine the peak, rms, and average values.
  3. What is the peak-to-peak ripple in the output voltage?
  4. Plot the currents through the switch, diode, and capacitor.

  ('$C', s_C),
  ('$N1', s_N1),
  ('$N2', s_N2),
  ('$f_hz', s_f_hz),
]
calc.replace_strings_1("flyback_1_orig.in", "flyback_1.in", l)
print('flyback_1.in is ready for execution')

import numpy as np
import matplotlib.pyplot as plt 
import gseim_calc as calc
from setsize import set_size

slv = calc.slv("flyback_1.in")

i_slv = 0
i_out = 0
filename = slv.l_filename_all[i_slv][i_out]
print('filename:', filename)
u = np.loadtxt(filename)
t = u[:, 0]

v_out = slv.get_array_double(i_slv,i_out,"v_out",u)
ILm   = slv.get_array_double(i_slv,i_out,"ILm"  ,u)
IS    = slv.get_array_double(i_slv,i_out,"IS"   ,u)
ID    = slv.get_array_double(i_slv,i_out,"ID"   ,u)
IC    = slv.get_array_double(i_slv,i_out,"IC"   ,u)

# since we have stored two cycles, we need to divide the last time point
# by 2 to get the period:

T = t[-1]/2

l_ILm = calc.avg_rms_3a(t, ILm, T, 2.0*T, 1.0e-4*T)

l_ILm_1 = calc.min_max_1(t, ILm, 0.0, 2.0*T)

print('rms inductor current:', "%11.4E"%l_ILm[1])
print('average inductor current:', "%11.4E"%l_ILm[0])
print('peak inductor current:', "%11.4E"%l_ILm_1[1])

l_v_out_1 = calc.min_max_1(t, v_out, 0.0, 2.0*T)
v_out_ptop = l_v_out_1[1] - l_v_out_1[0]
print('peak-to-peak ripple in Vo:', "%11.4E"%v_out_ptop)

color1='green'
color2='crimson'
color3='goldenrod'
color4='blue'
color5='cornflowerblue'

fig, ax = plt.subplots(5, sharex=False)
plt.subplots_adjust(wspace=0, hspace=0.0)

set_size(7, 7, ax[0])

for i in range(5):
    ax[i].set_xlim(left=0, right=2.0*T*1e6)
    ax[i].grid(color='#CCCCCC', linestyle='solid', linewidth=0.5)

ax[0].plot(t*1e6, ILm  , color=color1, linewidth=1.0, label="$i_{Lm}$")
ax[1].plot(t*1e6, IC   , color=color2, linewidth=1.0, label="$i_C$")
ax[2].plot(t*1e6, IS   , color=color3, linewidth=1.0, label="$i_S$")
ax[3].plot(t*1e6, ID   , color=color4, linewidth=1.0, label="$i_D$")
ax[4].plot(t*1e6, v_out, color=color5, linewidth=1.0, label="$V_o$")

ax[0].set_ylabel(r'$i_{Lm}$', fontsize=14)
ax[1].set_ylabel(r'$i_C$'   , fontsize=14)
ax[2].set_ylabel(r'$i_S$'   , fontsize=14)
ax[3].set_ylabel(r'$i_D$'   , fontsize=14)
ax[4].set_ylabel(r'$V_o$'   , fontsize=14)

ax[4].set_xlabel('time (' + r'$\mu$' + 'sec)', fontsize=14)
