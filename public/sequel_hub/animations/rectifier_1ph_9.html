import numpy as np
import matplotlib.pyplot as plt 
import gseim_calc as calc
from setsize import set_size

f_hz = 50.0
T = 1.0/f_hz

slv = calc.slv("rectifier_1ph_9.in")

i_slv = 0
i_out = 0
filename = slv.l_filename_all[i_slv][i_out]
print('filename:', filename)
u = np.loadtxt(filename)
t = u[:, 0]

ID1   = slv.get_array_double(i_slv,i_out,"ID1",u)
ID2   = slv.get_array_double(i_slv,i_out,"ID2",u)
ID3   = slv.get_array_double(i_slv,i_out,"ID3",u)
ID4   = slv.get_array_double(i_slv,i_out,"ID4",u)
V_VS1 = slv.get_array_double(i_slv,i_out,"V_VS1",u)
I_VS1 = slv.get_array_double(i_slv,i_out,"I_VS1",u)
I_R   = slv.get_array_double(i_slv,i_out,"I_R",u)
P_R   = slv.get_array_double(i_slv,i_out,"P_R",u)

I_R_avg = calc.avg_3(t, I_R, 0.0, 2.0*T, 1.0e-5*T)
P_R_avg = calc.avg_3(t, P_R, 0.0, 2.0*T, 1.0e-5*T)

print('average current through DC source:', "%11.4E"%(I_R_avg))
print('average power dissipated in R:', "%11.4E"%(P_R_avg))

# compute durations of diode conduction:

ndiv = 5000

delt_ID1, ID1p = calc.interp_linear_1(t, ID1, ndiv)
delt_ID2, ID2p = calc.interp_linear_1(t, ID2, ndiv)
delt_ID3, ID3p = calc.interp_linear_1(t, ID3, ndiv)
delt_ID4, ID4p = calc.interp_linear_1(t, ID4, ndiv)

n_ID1 = 0
n_ID2 = 0
n_ID3 = 0
n_ID4 = 0

for k in range(ndiv):
    if (ID1p[k] > 0): n_ID1 += 1
    if (ID2p[k] > 0): n_ID2 += 1
    if (ID3p[k] > 0): n_ID3 += 1
    if (ID4p[k] > 0): n_ID4 += 1

print('angle of conduction for D1:', "%6.2f"%(float(n_ID1)*delt_ID1*360.0/(2.0*T)), 'deg.')
print('angle of conduction for D2:', "%6.2f"%(float(n_ID1)*delt_ID2*360.0/(2.0*T)), 'deg.')
print('angle of conduction for D3:', "%6.2f"%(float(n_ID1)*delt_ID3*360.0/(2.0*T)), 'deg.')
print('angle of conduction for D4:', "%6.2f"%(float(n_ID1)*delt_ID4*360.0/(2.0*T)), 'deg.')

color1 = 'blue'
color2 = 'green'
color3 = 'red'
color4 = 'dodgerblue'
color5 = 'olive'

fig, ax = plt.subplots(4, sharex=False)
plt.subplots_adjust(wspace=0, hspace=0.0)

set_size(5.5, 8, ax[0])

for i in range(4):
    ax[i].set_xlim(left=0.0, right=2.0*T*1e3)
    ax[i].grid(color='#CCCCCC', linestyle='solid', linewidth=0.5)

ax[0].set_ylabel(r'$V_s$'   , fontsize=12)
ax[1].set_ylabel(r'$I_R$'   , fontsize=12)
ax[2].set_ylabel(r'$I_{D1}$', fontsize=12)
ax[3].set_ylabel(r'$I_{D2}$', fontsize=12)

for i in range(3):
    ax[i].tick_params(labelbottom=False)

ax[0].plot(t*1e3,  V_VS1, color=color1, linewidth=1.0, label="$V_{S1}$")
ax[1].plot(t*1e3,  I_R  , color=color2, linewidth=1.0, label="$I_R$")
ax[2].plot(t*1e3,  ID1  , color=color3, linewidth=1.0, label="$I_{D1}$")
ax[3].plot(t*1e3,  ID2  , color=color4, linewidth=1.0, label="$I_{D2}$")

ax[1].axhline(y=I_R_avg, color=color2, linewidth=1.0, label="$I_R^{avg}$", linestyle='--', dashes=(5,3))

ax[3].set_xlabel('time (msec)', fontsize=11)

ax[1].legend(loc = 'lower right',frameon = True, fontsize = 10, title = None,
  markerfirst = True, markerscale = 1.0, labelspacing = 0.5, columnspacing = 2.0,
  prop = {'size' : 12},)

#plt.tight_layout()
plt.show()
