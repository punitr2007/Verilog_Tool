Buck-Boost Converter: CCM

The buck-boost converter given below, operating at $25\,$kHz has input voltage of $40\,$V and has an output voltage of $40\,$V. The output is connected to a resistance of $20\,\Omega$ and $C=100\,\mu$F. The peak-to-peak ripple in inductor current is $2\,$A.
  1. Determine the duty ratio of the converter.
  2. What is the inductance $L$?
  3. Plot the inductor current and capacitor voltage and mark the peaks.
  4. Find the avarage current through the inductor, switch, and diode.
  5. Find the peak-to-peak ripple voltage across $C$.
  6. Find the RMS current through inductor, switch, diode and capacitor.

  ('$Vin', s_Vin),
  ('$L', s_L),
  ('$C', s_C),
  ('$R', s_R),
  ('$D', s_D),
  ('$f_hz', s_f_hz)
]
calc.replace_strings_1("buck_boost_ccm_1_orig.in", "buck_boost_ccm_1.in", l)
print('buck_boost_ccm_1.in is ready for execution')

import numpy as np
import matplotlib.pyplot as plt 
import gseim_calc as calc
from setsize import set_size

slv = calc.slv("buck_boost_ccm_1.in")

i_slv = 0
i_out = 0
filename = slv.l_filename_all[i_slv][i_out]
print('filename:', filename)
u = np.loadtxt(filename)
t1 = u[:, 0]
t = t1*1e6 # convert time to micro-seconds

IL = slv.get_array_double(i_slv,i_out,"IL",u)
IS = slv.get_array_double(i_slv,i_out,"IS",u)
ID = slv.get_array_double(i_slv,i_out,"ID",u)
IC = slv.get_array_double(i_slv,i_out,"IC",u)
v_out = slv.get_array_double(i_slv,i_out,"v_out",u)
clock = slv.get_array_double(i_slv,i_out,"clock",u)

# since we have stored two cycles, we need to divide the last time point
# by 2 to get the period:

T = t[-1]/2

l_IL = calc.avg_rms_3a(t, IL, 0.0, 2.0*T, 1.0e-3*T)
l_IS = calc.avg_rms_3a(t, IS, 0.0, 2.0*T, 1.0e-3*T)
l_ID = calc.avg_rms_3a(t, ID, 0.0, 2.0*T, 1.0e-3*T)
l_IC = calc.avg_rms_3a(t, IC, 0.0, 2.0*T, 1.0e-3*T)
l_v_out = calc.avg_rms_3a(t, v_out, 0.0, 2.0*T, 1.0e-3*T)

l1 = calc.min_max_1(t, IL, 0.0, 2.0*T)
l2 = calc.min_max_1(t, v_out, 0.0, 2.0*T)

print('average output voltage:', "%11.4E"%l_v_out[0])
print('average inductor current:', "%11.4E"%l_IL[0])
print('average switch current:', "%11.4E"%l_IS[0])
print('average diode current:', "%11.4E"%l_ID[0])
print('peak-to-peak capacitor voltage:', "%11.4E"%(l2[1]-l2[0]))
print('rms inductor current:', "%11.4E"%l_IL[1])
print('rms switch current:', "%11.4E"%l_IS[1])
print('rms diode current:', "%11.4E"%l_ID[1])
print('rms capacitor current:', "%11.4E"%l_IC[1])

N = 10
t1 = np.linspace(0.0, 2.0*T, N)
ILmin = np.array([l1[0]]*N)
ILmax = np.array([l1[1]]*N)

color1='green'
color2='crimson'
color3='goldenrod'
color4='blue'

fig, ax = plt.subplots(2, sharex=False)
plt.subplots_adjust(wspace=0, hspace=0.0)

set_size(4, 5, ax[0])

ax[0].grid(color='#CCCCCC', linestyle='solid', linewidth=0.5)
ax[0].set_xlim(left=0.0, right=2.0*T)
ax[0].set_xlabel('time (' + r'$\mu$' + 'sec)', fontsize=11)
ax[0].plot(t, IL, color=color1, linewidth=1.0, label="$i_L$")
ax[0].plot(t1, ILmin, color=color1, linewidth=1.0, label="$i_L^{min}$", linestyle="--", dashes=(5,3))
ax[0].plot(t1, ILmax, color=color1, linewidth=1.0, label="$i_L^{max}$", linestyle="-.")
ax[0].axhline(y=l_IL[0], color=color1, linewidth=1.0, label="$i_L^{avg}$", linestyle="dotted")
ax[0].legend(loc = 'lower right',frameon = True, fontsize = 10, title = None,
   markerfirst = True, markerscale = 1.0, labelspacing = 0.5, columnspacing = 2.0,
   prop = {'size' : 12},)

ax[1].grid(color='#CCCCCC', linestyle='solid', linewidth=0.5)
ax[1].set_xlim(left=0.0, right=2.0*T)
ax[1].set_xlabel('time (' + r'$\mu$' + 'sec)', fontsize=11)
ax[1].plot(t, v_out, color=color2, linewidth=1.0, label="$V_o$")
ax[1].axhline(y=l_v_out[0], color=color2, linewidth=1.0, label="$V_o^{avg}$", linestyle="dotted")
ax[1].legend(loc = 'lower right',frameon = True, fontsize = 10, title = None,
   markerfirst = True, markerscale = 1.0, labelspacing = 0.5, columnspacing = 2.0,
   prop = {'size' : 12},)

plt.tight_layout()
plt.show()
