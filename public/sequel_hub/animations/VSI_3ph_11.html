We now replace the strings such as \$Vdc, \$L, with the values of our choice by running the python script given below. It takes an existing circuit file VSI_3ph_11_orig.in and produces a new circuit file VSI_3ph_11.in, after replacing \$Vdcby2, \$L, etc. with values of our choice.

import gseim_calc as calc
s_Vdcby2 = "300"
s_R = "10"
s_f_clock = "50"

# 180-deg conduction:
s_D = "0.5"

# 120-deg conduction:
#s_D = "0.33333"

l = [
  ('$Vdcby2', s_Vdcby2),
  ('$R', s_R),
  ('$D', s_D),
  ('$f_clock', s_f_clock),
]
calc.replace_strings_1("VSI_3ph_11_orig.in", "VSI_3ph_11.in", l)
print('VSI_3ph_11.in is ready for execution')

import numpy as np
import matplotlib.pyplot as plt 
import gseim_calc as calc
from setsize import set_size

f_hz = 50.0
T = 1.0/f_hz

slv = calc.slv("VSI_3ph_11.in")

i_slv = 0
i_out = 0
filename = slv.l_filename_all[i_slv][i_out]
print('filename:', filename)
u1 = np.loadtxt(filename)
t1 = u1[:, 0]

g1 = slv.get_array_double(i_slv,i_out,"g1",u1)
g2 = slv.get_array_double(i_slv,i_out,"g2",u1)
g3 = slv.get_array_double(i_slv,i_out,"g3",u1)
g4 = slv.get_array_double(i_slv,i_out,"g4",u1)
g5 = slv.get_array_double(i_slv,i_out,"g5",u1)
g6 = slv.get_array_double(i_slv,i_out,"g6",u1)

i_out = 1
filename = slv.l_filename_all[i_slv][i_out]
print('filename:', filename)
u2 = np.loadtxt(filename)
t2 = u2[:, 0]

v_an = slv.get_array_double(i_slv,i_out,"v_an",u2)
v_a0 = slv.get_array_double(i_slv,i_out,"v_a0",u2)
v_n0 = slv.get_array_double(i_slv,i_out,"v_n0",u2)

fig, ax = plt.subplots(4, sharex=False, gridspec_kw={'height_ratios': [2, 1, 1, 1]})
plt.subplots_adjust(wspace=0, hspace=0.0)

set_size(6.5, 7, ax[0])

for i in range(4):
    ax[i].set_xlim(left=0.0, right=2.0*T*1e3)
    ax[i].grid(color='#CCCCCC', linestyle='solid', linewidth=0.5)

ax[0].set_ylabel(r'$g_x$'   , fontsize=12)
ax[1].set_ylabel(r'$v_{an}$', fontsize=12)
ax[2].set_ylabel(r'$v_{a0}$', fontsize=12)
ax[3].set_ylabel(r'$v_{n0}$', fontsize=12)

for i in range(3):
    ax[i].tick_params(labelbottom=False)

color1 = "tomato"
color2 = "dodgerblue"
color3 = "olive"
color4 = "blue"
color5 = "grey"
color6 = "green"
color7 = "crimson"

dy = 1.5

ax[0].plot(t1*1e3, (g1       ), color=color1, linewidth=1.0, label="$g_1$")
ax[0].plot(t1*1e3, (g2 -   dy), color=color2, linewidth=1.0, label="$g_2$")
ax[0].plot(t1*1e3, (g3 - 2*dy), color=color3, linewidth=1.0, label="$g_3$")
ax[0].plot(t1*1e3, (g4 - 3*dy), color=color4, linewidth=1.0, label="$g_4$")
ax[0].plot(t1*1e3, (g5 - 4*dy), color=color5, linewidth=1.0, label="$g_5$")
ax[0].plot(t1*1e3, (g6 - 5*dy), color=color6, linewidth=1.0, label="$g_6$")

ax[0].tick_params(left = False)
ax[0].set_yticks([])
