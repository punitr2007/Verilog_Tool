Forward converter

The forward converter shown in the figure has an input voltage of $20\,$V and an output voltage of $40\,$V. The switching frequency of the converter is $100\,$kHz. The turns ratio of the transformer is 1:1:4. The magnetizing inductance of the transformer as seen from the source side is $1\,$mH. The load resistance is $10\,\Omega$, the output capacitance is $10\,\mu$F, and the filter inductance is $1\,$mH.
  1. What is the duty cycle of the converter?
  2. Plot the current through $L$, and determine the peak, rms, and average values.
  3. What is the peak-to-peak ripple in the output voltage?
  4. Plot the current through the source and determine its peak value.
  5. Plot the currents through the switch $S$ and diode $D_1$.

import numpy as np
import matplotlib.pyplot as plt 
import gseim_calc as calc
from setsize import set_size

slv = calc.slv("forward_1.in")

i_slv = 0
i_out = 0
filename = slv.l_filename_all[i_slv][i_out]
print('filename:', filename)
u = np.loadtxt(filename)
t = u[:, 0]

v_out = slv.get_array_double(i_slv,i_out,"v_out",u)
IL    = slv.get_array_double(i_slv,i_out,"IL"   ,u)
ILm   = slv.get_array_double(i_slv,i_out,"ILm"  ,u)
IS    = slv.get_array_double(i_slv,i_out,"IS"   ,u)
ID1   = slv.get_array_double(i_slv,i_out,"ID1"  ,u)
IVs   = slv.get_array_double(i_slv,i_out,"IVs"  ,u)

# since we have stored two cycles, we need to divide the last time point
# by 2 to get the period:

T = t[-1]/2

l_IL = calc.avg_rms_3a(t, IL, T, 2.0*T, 1.0e-4*T)

l_IL_1 = calc.min_max_1(t, IL, 0.0, 2.0*T)

print('rms inductor current:', "%11.4E"%l_IL[1], "A")
print('average inductor current:', "%11.4E"%l_IL[0], "A")
print('peak inductor current:', "%11.4E"%l_IL_1[1], "A")

l_v_out_1 = calc.min_max_1(t, v_out, 0.0, 2.0*T)
v_out_ptop = l_v_out_1[1] - l_v_out_1[0]
print('peak-to-peak ripple in Vo:', "%11.4E"%v_out_ptop, "V")

l_IVs_1 = calc.min_max_1(t, IVs, 0.0, 2.0*T)
IVs_max = l_IVs_1[1]
print('peak source current:', "%11.4E"%IVs_max, "A")

color1='green'
color2='crimson'
color3='goldenrod'
color4='blue'
color5='cornflowerblue'
color6='red'

fig, ax = plt.subplots(6, sharex=False)
plt.subplots_adjust(wspace=0, hspace=0.0)

set_size(7, 7, ax[0])

for i in range(6):
    ax[i].set_xlim(left=0, right=2.0*T*1e6)
    ax[i].grid(color='#CCCCCC', linestyle='solid', linewidth=0.5)

ax[0].plot(t*1e6, IL   , color=color1, linewidth=1.0, label="$i_{Lm}$")
ax[1].plot(t*1e6, IVs  , color=color2, linewidth=1.0, label="$i_{Vs}$")
ax[2].plot(t*1e6, IS   , color=color3, linewidth=1.0, label="$i_S$")
ax[3].plot(t*1e6, ID1  , color=color4, linewidth=1.0, label="$i_{D1}$")
ax[4].plot(t*1e6, v_out, color=color5, linewidth=1.0, label="$V_o$")
ax[5].plot(t*1e6, ILm  , color=color5, linewidth=1.0, label="$i_{Lm}$")

ax[0].set_ylabel(r'$i_L$'   , fontsize=14)
ax[1].set_ylabel(r'$i_{Vs}$', fontsize=14)
ax[2].set_ylabel(r'$i_S$'   , fontsize=14)
ax[3].set_ylabel(r'$i_{D1}$', fontsize=14)
ax[4].set_ylabel(r'$V_o$'   , fontsize=14)
ax[5].set_ylabel(r'$i_{Lm}$', fontsize=14)

ax[5].set_xlabel('time (' + r'$\mu$' + 'sec)', fontsize=14)

for i in range(5):
    ax[i].tick_params(labelbottom=False)

#plt.tight_layout()
plt.show()
