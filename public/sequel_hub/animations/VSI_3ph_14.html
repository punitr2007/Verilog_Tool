import gseim_calc as calc
s_Vdc = "600"
s_f_clock = "50"
s_L = "10e-3"

f_clock = float(s_f_clock)
T = 1/f_clock
s_2T = ("%11.4E"%(2.0*T)).strip()

phi = 180.0
D1 = phi/360.0
s_D1 = ("%11.4E"%(D1)).strip()

l = [
  ('$Vdc', s_Vdc),
  ('$L', s_L),
  ('$f_clock', s_f_clock),
  ('$D1', s_D1),
  ('$2T', s_2T),
]
calc.replace_strings_1("VSI_3ph_14_orig.in", "VSI_3ph_14.in", l)
print('VSI_3ph_14.in is ready for execution')
