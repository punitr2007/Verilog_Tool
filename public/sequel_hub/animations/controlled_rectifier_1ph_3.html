import gseim_calc as calc
import numpy as np

s_R = "10"
s_L = "10e-3"
s_alpha = "30"

l = [
  ('$R', s_R),
  ('$L', s_L),
  ('$alpha', s_alpha),
]
calc.replace_strings_1("controlled_rectifier_1ph_3_orig.in", "controlled_rectifier_1ph_3.in", l)
print('controlled_rectifier_1ph_3.in is ready for execution')

import numpy as np
import matplotlib.pyplot as plt 
import gseim_calc as calc
from setsize import set_size

f_hz = 50.0
T = 1.0/f_hz

slv = calc.slv("controlled_rectifier_1ph_3_orig.in")

i_slv = 0
i_out = 0
filename = slv.l_filename_all[i_slv][i_out]
print('filename:', filename)
u = np.loadtxt(filename)
t = u[:, 0]

v_s = slv.get_array_double(i_slv,i_out,"v_s",u)
v_o = slv.get_array_double(i_slv,i_out,"v_o",u)
g   = slv.get_array_double(i_slv,i_out,"g",u)
IT   = slv.get_array_double(i_slv,i_out,"IT",u)
ID   = slv.get_array_double(i_slv,i_out,"ID",u)
IR   = slv.get_array_double(i_slv,i_out,"IR",u)

l_IR = calc.avg_rms_3a(t, IR, 0.0, 2.0*T, 1.0e-5*T)

print('average load current:', "%11.4E"%l_IR[0])

color1='blue'
color2='green'
color3='red'
color4='dodgerblue'

fig, ax = plt.subplots(6, sharex=False)
plt.subplots_adjust(wspace=0, hspace=0.0)

set_size(5.5, 11, ax[0])

for i in range(6):
    ax[i].set_xlim(left=0.0, right=2.0*T*1e3)
    ax[i].grid(color='#CCCCCC', linestyle='solid', linewidth=0.5)

ax[0].set_ylabel(r'$v_s$', fontsize=12)
ax[1].set_ylabel(r'$g$'  , fontsize=12)
ax[2].set_ylabel(r'$v_o$', fontsize=12)
ax[3].set_ylabel(r'$I_T$', fontsize=12)
ax[4].set_ylabel(r'$I_D$', fontsize=12)
ax[5].set_ylabel(r'$I_R$', fontsize=12)

ax[1].set_yticks([0.0, 1.0])

for k in range(5):
    ax[k].tick_params(labelbottom=False)

ax[0].plot(t*1e3, v_s, color=color1, linewidth=1.0, label="$v_s$")
ax[1].plot(t*1e3, g  , color=color2, linewidth=1.0, label="$g$")
ax[2].plot(t*1e3, v_o, color=color3, linewidth=1.0, label="$v_o$")
ax[3].plot(t*1e3, IT , color=color4, linewidth=1.0, label="$I_T$")
ax[4].plot(t*1e3, ID , color=color4, linewidth=1.0, label="$I_D$")
ax[5].plot(t*1e3, IR , color=color4, linewidth=1.0, label="$I_R$")

ax[5].set_xlabel('time (msec)', fontsize=12)

#plt.tight_layout()
plt.show()
