import gseim_calc as calc
import numpy as np

s_L0 = "1"
s_L1 = "10e-3"
s_alpha = "45" # to be changed by user
s_Vdc = "100"

l = [
  ('$L0', s_L0),
  ('$L1', s_L1),
  ('$alpha', s_alpha),
  ('$Vdc', s_Vdc),
  ]
calc.replace_strings_1("controlled_rectifier_1ph_6_orig.in", "controlled_rectifier_1ph_6.in", l)
print('controlled_rectifier_1ph_6.in is ready for execution')

import numpy as np
import matplotlib.pyplot as plt 
import gseim_calc as calc
from setsize import set_size

f_hz = 50.0
T = 1.0/f_hz

slv = calc.slv("controlled_rectifier_1ph_6_orig.in")

i_slv = 0
i_out = 0
filename = slv.l_filename_all[i_slv][i_out]
print('filename:', filename)
u = np.loadtxt(filename)
t = u[:, 0]

v_s1 = slv.get_array_double(i_slv,i_out,"v_s1",u)
g1   = slv.get_array_double(i_slv,i_out,"g1",u)
g2   = slv.get_array_double(i_slv,i_out,"g2",u)
IT1  = slv.get_array_double(i_slv,i_out,"IT1",u)
IT2  = slv.get_array_double(i_slv,i_out,"IT2",u)
IT3  = slv.get_array_double(i_slv,i_out,"IT3",u)
IT4  = slv.get_array_double(i_slv,i_out,"IT4",u)
IL   = slv.get_array_double(i_slv,i_out,"IL",u)
IL1  = slv.get_array_double(i_slv,i_out,"IL1",u)

l_IL = calc.avg_rms_3a(t, IL, 0.0, 2.0*T, 1.0e-5*T)

print('average value of Id:', "%11.4E"%l_IL[0])

color1='blue'
color2='green'
color3='red'
color4='dodgerblue'

fig, ax = plt.subplots(7, sharex=False)
plt.subplots_adjust(wspace=0, hspace=0.0)

set_size(5.5, 12, ax[0])

for i in range(7):
    ax[i].set_xlim(left=0.0, right=2.0*T*1e3)
    ax[i].grid(color='#CCCCCC', linestyle='solid', linewidth=0.5)

ax[0].set_ylabel(r'$v_{s1}$', fontsize=12)
ax[1].set_ylabel(r'$g_1$'   , fontsize=12)
ax[2].set_ylabel(r'$g_2$'   , fontsize=12)
ax[3].set_ylabel(r'$I_{T1}$', fontsize=12)
ax[4].set_ylabel(r'$I_{T2}$', fontsize=12)
ax[5].set_ylabel(r'$I_d$'   , fontsize=12)
ax[6].set_ylabel(r'$I_{L1}$', fontsize=12)

ax[1].set_yticks([0.0, 1.0])
ax[2].set_yticks([0.0, 1.0])

for k in range(6):
    ax[k].tick_params(labelbottom=False)

ax[0].plot(t*1e3, v_s1, color=color1, linewidth=1.0, label="$v_{s1}$")
ax[1].plot(t*1e3, g1  , color=color2, linewidth=1.0, label="$g_1$")
ax[2].plot(t*1e3, g2  , color=color2, linewidth=1.0, label="$g_2$")
ax[3].plot(t*1e3, IT1 , color=color4, linewidth=1.0, label="$I_{T1}$")
ax[4].plot(t*1e3, IT2 , color=color4, linewidth=1.0, label="$I_{T2}$")
ax[5].plot(t*1e3, IL  , color=color4, linewidth=1.0, label="$I_d$")
ax[6].plot(t*1e3, IL1 , color=color4, linewidth=1.0, label="$I_{L1}$")

ax[5].axhline(y=l_IL[0], color=color4, linewidth=1.0, label="$I_d^{avg}$", linestyle='--', dashes=(5,3))

ax[6].set_xlabel('time (msec)', fontsize=12)

ax[5].legend(loc = 'upper right',frameon = True, fontsize = 10, title = None,
  markerfirst = True, markerscale = 1.0, labelspacing = 0.5, columnspacing = 2.0,
  prop = {'size' : 12},)

#plt.tight_layout()
plt.show()
