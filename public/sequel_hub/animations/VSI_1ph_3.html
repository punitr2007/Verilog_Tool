1-phase VSI

In the inverter circuit, $V_{S1}$ is a square wave source with amplitude $300\,$V. The phase of $V_{S1}$ is the same as that of the output voltage $V_{BC}$ of the inverter. The output of the inverter is a square wave with frequency of $50\,$Hz, and the DC link voltage $V_{DC}=400\,$V. The other parameters are $R=2\,\Omega$ and $L=20\,$mH.
  1. Plot the currents $i_{dc}$ and $i_L$.
  2. Find the average power delivered to $V_{S1}$.
  3. Find the power delivered to the load.
  4. Find the average power transferred from the DC voltage source.
  5. Find the peak values of $i_{dc}$ and $i_L$.

import gseim_calc as calc
s_Vdc = '400'
s_R = '2'
s_L = '20e-3'

f_hz = 50.0
s_f_hz = ("%11.4E"%(f_hz)).strip()

T = 1/f_hz
s_Tby2 = ("%11.4E"%(T/2)).strip()

VS1_peak = 300.0

s_L1 = ("%11.4E"%VS1_peak).strip()
s_L2 = ("%11.4E"%(-VS1_peak)).strip()

l = [
  ('$Vdc', s_Vdc),
  ('$R', s_R),
  ('$L', s_L),
  ('$f_hz', s_f_hz),
  ('$Tby2', s_Tby2),
  ('$VS1_L1', s_L1),
  ('$VS1_L2', s_L2)
]
calc.replace_strings_1("VSI_1ph_3_orig.in", "VSI_1ph_3.in", l)
print('VSI_1ph_3.in is ready for execution')

import numpy as np
import matplotlib.pyplot as plt 
import gseim_calc as calc
from setsize import set_size

slv = calc.slv("VSI_1ph_3.in")

i_slv = 0
i_out = 0
filename = slv.l_filename_all[i_slv][i_out]
print('filename:', filename)
u = np.loadtxt(filename)
t1 = u[:, 0]
t = 1e3*t1 # convert time to msec

v_out = slv.get_array_double(i_slv,i_out,"v_out",u)
IR    = slv.get_array_double(i_slv,i_out,"IR"   ,u)
Idc   = slv.get_array_double(i_slv,i_out,"Idc"  ,u)
P_R   = slv.get_array_double(i_slv,i_out,"P_R"  ,u)
P_dc  = slv.get_array_double(i_slv,i_out,"P_dc" ,u)
P_S1  = slv.get_array_double(i_slv,i_out,"P_S1" ,u)
g1    = slv.get_array_double(i_slv,i_out,"g1"   ,u)
g2    = slv.get_array_double(i_slv,i_out,"g2"   ,u)

# since we have stored two cycles, we need to divide the last time point
# by 2 to get the period:

T = t[-1]/2

l_IR   = calc.avg_rms_3a(t, IR  , 0.0, 2.0*T, 1.0e-4*T)
l_P_R  = calc.avg_rms_3a(t, P_R , 0.0, 2.0*T, 1.0e-4*T)
l_P_dc = calc.avg_rms_3a(t, P_dc, 0.0, 2.0*T, 1.0e-4*T)
l_P_S1 = calc.avg_rms_3a(t, P_S1, 0.0, 2.0*T, 1.0e-4*T)
l_Idc  = calc.avg_rms_3a(t, Idc , 0.0, 2.0*T, 1.0e-4*T)

l_IR_1  = calc.min_max_1(t, IR , 0.0, 2.0*T)
l_Idc_1 = calc.min_max_1(t, Idc, 0.0, 2.0*T)

print('average DC source current:', "%11.4E"%l_Idc[0])
print('rms DC source current:', "%11.4E"%l_Idc[1])
print('rms load current:', "%11.4E"%l_IR[1])
print('peak load current:', "%11.4E"%l_IR_1[1])
print('power absorbed by load:', "%11.4E"%l_P_R[0])
print('power delivered by Vdc:', "%11.4E"%l_P_dc[0])
print('power delivered by VS1:', "%11.4E"%l_P_S1[0])
print('peak Idc:', l_Idc_1[1])
print('peak IL:', l_IR_1[1])
