import numpy as np
import matplotlib.pyplot as plt 
import gseim_calc as calc
from setsize import set_size

f_hz = 50.0
T = 1.0/f_hz

slv = calc.slv("rectifier_1ph_11.in")

i_slv = 0
i_out = 0
filename = slv.l_filename_all[i_slv][i_out]
print('filename:', filename)
u = np.loadtxt(filename)
t = u[:, 0]

ID1   = slv.get_array_double(i_slv,i_out,"ID1",u)
ID2   = slv.get_array_double(i_slv,i_out,"ID2",u)
ID3   = slv.get_array_double(i_slv,i_out,"ID3",u)
ID4   = slv.get_array_double(i_slv,i_out,"ID4",u)
IL    = slv.get_array_double(i_slv,i_out,"IL",u)
v_s   = slv.get_array_double(i_slv,i_out,"v_s",u)
v_o   = slv.get_array_double(i_slv,i_out,"v_o",u)

l_v_o = calc.avg_rms_3a(t, v_o, 0.0, 2.0*T, 1.0e-5*T)

print('average voltage across current source:', "%11.4E"%(l_v_o[0]))

# compute theta:

ndiv = 5000

delt_v_o, v_op = calc.interp_linear_1(t, v_o, ndiv)

n_v_o = 0

for k in range(ndiv):
    if (v_op[k] < 1.0e-10): n_v_o += 1

theta = 0.5*float(n_v_o)*delt_v_o*360.0/(2.0*T)
print('theta:', theta, 'deg')

color1 = 'blue'
color2 = 'green'
color3 = 'red'
color4 = 'dodgerblue'
color5 = 'olive'

fig, ax = plt.subplots(4, sharex=False)
plt.subplots_adjust(wspace=0, hspace=0.0)

set_size(5.5, 8, ax[0])

for i in range(4):
    ax[i].set_xlim(left=0.0, right=2.0*T*1e3)
    ax[i].grid(color='#CCCCCC', linestyle='solid', linewidth=0.5)

ax[0].set_ylabel(r'$v_s$'      , fontsize=12)
ax[1].set_ylabel(r'$v_o$'      , fontsize=12)
ax[2].set_ylabel(r'$I_{diode}$', fontsize=12)
ax[3].set_ylabel(r'$I_L$'      , fontsize=12)

for i in range(3):
    ax[i].tick_params(labelbottom=False)

ax[0].plot(t*1e3,  v_s, color=color1, linewidth=1.0, label="$v_s$")
ax[1].plot(t*1e3,  v_o, color=color2, linewidth=1.0, label="$v_o$")
ax[2].plot(t*1e3,  ID1, color=color3, linewidth=1.0, label="$I_{D1}$")
ax[2].plot(t*1e3,  ID2, color=color4, linewidth=1.0, label="$I_{D2}$")
ax[3].plot(t*1e3,  IL , color=color5, linewidth=1.0, label="$I_L$")

ax[1].axhline(l_v_o[0], color=color2, linewidth=1.0, label="$v_o^{avg}$", linestyle='--', dashes=(5,3))

ax[3].set_xlabel('time (msec)', fontsize=11)

ax[2].legend(loc = 'lower right',frameon = True, fontsize = 10, title = None,
  markerfirst = True, markerscale = 1.0, labelspacing = 0.5, columnspacing = 2.0,
  prop = {'size' : 12},)

#plt.tight_layout()
plt.show()
