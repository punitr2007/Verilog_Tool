Two-switch forward converter

In the two-switch forward converter shown in the figure, the duty ratio is 0.4, and the switching frequency is $20\,$kHz. The input voltage of the converter is $100\,$V, and it is connected to a load resistance of $4\,\Omega$. The output voltage of the converter is $20\,$V. The magnetizing inductance of the transformer as seen from the source side is $5\,$mH. The output capacitance is $20\,\mu$F, and the filter inductance is $500\,\mu$H.
  1. Determine the turns ratio of the transformer.
  2. Determine the peak magnetizing current of the transformer.
  3. What is the duration of conduction for $D_1$ and $D_2$?
  4. Plot the current drawn from the DC source and determine its peak.

import gseim_calc as calc
s_Vdc = "100"
s_Lm = "5e-3"
s_L = "500e-6"
s_R = "4"
s_C = "20e-6"
s_N1 = "1"
s_N2 = "0.5" # to be entered by user

T = 1/f_hz
s_Tend = "4e-3"
Tend = float(s_Tend)
T1 = Tend - 2.0*T
s_T1 = ("%11.4E"%(T1)).strip()

l = [
  ('$Vdc', s_Vdc),
  ('$Lm', s_Lm),
  ('$L', s_L),
  ('$R', s_R),
  ('$C', s_C),
  ('$N1', s_N1),
  ('$N2', s_N2),
  ('$Tend', s_Tend),
  ('$T1', s_T1),
]
calc.replace_strings_1("forward_3_orig.in", "forward_3.in", l)
print('forward_3.in is ready for execution')

import numpy as np
import matplotlib.pyplot as plt 
import gseim_calc as calc
from setsize import set_size

slv = calc.slv("forward_3.in")

i_slv = 0
i_out = 0
filename = slv.l_filename_all[i_slv][i_out]
print('filename:', filename)
u = np.loadtxt(filename)
ta = u[:, 0]
t = ta -ta[0]

v_out = slv.get_array_double(i_slv,i_out,"v_out",u)
IL    = slv.get_array_double(i_slv,i_out,"IL"   ,u)
ILm   = slv.get_array_double(i_slv,i_out,"ILm"  ,u)
ID1   = slv.get_array_double(i_slv,i_out,"ID1"  ,u)
ID2   = slv.get_array_double(i_slv,i_out,"ID2"  ,u)
IVs   = slv.get_array_double(i_slv,i_out,"IVs"  ,u)

# since we have stored two cycles, we need to divide the last time point
# by 2 to get the period:

T = t[-1]/2

l_ILm_1 = calc.min_max_1(t, ILm, 0.0, 2.0*T)
print('maximum value of ILm:', "%11.4E"%l_ILm_1[1], " A")

l_IVs_1 = calc.min_max_1(t, IVs, 0.0, 2.0*T)
print('maximum value of IVs:', "%11.4E"%l_IVs_1[1], " A")

ndiv = 5000
n_ID1 = 0
n_ID2 = 0

delt_ID1, ID1p = calc.interp_linear_1(t, ID1, ndiv)
delt_ID2, ID2p = calc.interp_linear_1(t, ID2, ndiv)

for k in range(ndiv):
    if (ID1p[k] > 0): n_ID1 += 1
    if (ID2p[k] > 0): n_ID2 += 1

print('duration of conduction for D1 in one cycle:',
  "%11.4E"%(1.0e6*float(n_ID1)*delt_ID1/(2.0)), 'micro seconds.')
print('duration of conduction for D2 in one cycle:',
  "%11.4E"%(1.0e6*float(n_ID2)*delt_ID2/(2.0)), 'micro seconds.')

color1='green'
color2='crimson'
color3='goldenrod'
color4='blue'
color5='cornflowerblue'
color6='red'

fig, ax = plt.subplots(6, sharex=False)
plt.subplots_adjust(wspace=0, hspace=0.0)

set_size(7, 7, ax[0])

for i in range(6):
    ax[i].set_xlim(left=0, right=2.0*T*1e6)
    ax[i].grid(color='#CCCCCC', linestyle='solid', linewidth=0.5)

ax[0].plot(t*1e6, IL   , color=color1, linewidth=1.0, label="$i_L$")
ax[1].plot(t*1e6, IVs  , color=color2, linewidth=1.0, label="$i_{Vs}$")
ax[2].plot(t*1e6, ID1  , color=color3, linewidth=1.0, label="$i_{D1}$")
ax[3].plot(t*1e6, ID2  , color=color4, linewidth=1.0, label="$i_{D2}$")
ax[4].plot(t*1e6, v_out, color=color5, linewidth=1.0, label="$V_o$")
ax[5].plot(t*1e6, ILm  , color=color6, linewidth=1.0, label="$i_{Lm}$")

ax[0].set_ylabel(r'$i_L$'   , fontsize=14)
ax[1].set_ylabel(r'$i_{Vs}$', fontsize=14)
ax[2].set_ylabel(r'$i_{D1}$', fontsize=14)
ax[3].set_ylabel(r'$i_{D2}$', fontsize=14)
ax[4].set_ylabel(r'$V_o$'   , fontsize=14)
ax[5].set_ylabel(r'$i_{Lm}$', fontsize=14)

ax[5].set_xlabel('time (' + r'$\mu$' + 'sec)', fontsize=14)

for i in range(5):
    ax[i].tick_params(labelbottom=False)

#plt.tight_layout()
plt.show()
