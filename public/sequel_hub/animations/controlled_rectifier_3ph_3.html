Determine the minimum firing angle $\alpha$ at which the freewheeling diode D starts to conduct.

import gseim_calc as calc
import numpy as np

s_alpha = "90" # to be changed by user

VL = 400.0
A_sin = VL*np.sqrt(2/3)

s_A_sin = ("%11.4E"%(A_sin)).strip()

l = [
  ('$alpha', s_alpha),
  ('$A_sin', s_A_sin),
]
calc.replace_strings_1("controlled_rectifier_3ph_3_orig.in", "controlled_rectifier_3ph_3.in", l)
print('controlled_rectifier_3ph_3.in is ready for execution')
