3-phase CSI

A three-phase current source inverter as shown in the figure is feeding a star-connected resistive load. The circuit parameters are $I_s = 10\,$A, $R_a = 30\,\Omega$. What is the power consumed by the three-phase load for 180-degree and 120-degree conduction?

import gseim_calc as calc
s_Idc = "10"
s_R = "30"
s_f_clock = "50"

# 180-deg conduction:
s_D = "0.5"

# 120-deg conduction:
#s_D = "0.33333"

l = [
  ('$Idc', s_Idc),
  ('$R', s_R),
  ('$D', s_D),
  ('$f_clock', s_f_clock),
]
calc.replace_strings_1("CSI_3ph_1_orig.in", "CSI_3ph_1.in", l)
print('CSI_3ph_1.in is ready for execution')

import numpy as np
import matplotlib.pyplot as plt 
import gseim_calc as calc
from setsize import set_size

f_hz = 50.0
T = 1.0/f_hz

slv = calc.slv("CSI_3ph_1.in")

i_slv = 0
i_out = 0
filename = slv.l_filename_all[i_slv][i_out]
print('filename:', filename)
u1 = np.loadtxt(filename)
t1 = u1[:, 0]

g1 = slv.get_array_double(i_slv,i_out,"g1", u1)
g2 = slv.get_array_double(i_slv,i_out,"g2", u1)
g3 = slv.get_array_double(i_slv,i_out,"g3", u1)
g4 = slv.get_array_double(i_slv,i_out,"g4", u1)
g5 = slv.get_array_double(i_slv,i_out,"g5", u1)
g6 = slv.get_array_double(i_slv,i_out,"g6", u1)

i_out = 1
filename = slv.l_filename_all[i_slv][i_out]
print('filename:', filename)
u2 = np.loadtxt(filename)
t2 = u2[:, 0]

p_Ra = slv.get_array_double(i_slv,i_out,"p_Ra", u2)
p_Rb = slv.get_array_double(i_slv,i_out,"p_Rb", u2)
p_Rc = slv.get_array_double(i_slv,i_out,"p_Rc", u2)

i_out = 2
filename = slv.l_filename_all[i_slv][i_out]
print('filename:', filename)
u3 = np.loadtxt(filename)
t3 = u3[:, 0]

i_a = slv.get_array_double(i_slv,i_out,"i_a", u3)

l_p_Ra = calc.avg_rms_3a(t2, p_Ra, T, 2.0*T, 1.0e-4*T)
l_p_Rb = calc.avg_rms_3a(t2, p_Rb, T, 2.0*T, 1.0e-4*T)
l_p_Rc = calc.avg_rms_3a(t2, p_Rc, T, 2.0*T, 1.0e-4*T)

pwr_total = l_p_Ra[0] + l_p_Rb[0] + l_p_Rc[0]
print('average power delivered to load:', "%11.4E"%pwr_total)

fig, ax = plt.subplots(2, sharex=False, gridspec_kw={'height_ratios': [2, 1]})
plt.subplots_adjust(wspace=0, hspace=0.0)

set_size(6.5, 4, ax[0])

for i in range(2):
    ax[i].set_xlim(left=0.0, right=2.0*T*1e3)
    ax[i].grid(color='#CCCCCC', linestyle='solid', linewidth=0.5)

ax[0].set_ylabel(r'$g_x$', fontsize=12)
ax[1].set_ylabel(r'$i_a$', fontsize=12)

ax[0].tick_params(labelbottom=False)

color1 = "tomato"
color2 = "dodgerblue"
color3 = "olive"
color4 = "blue"
color5 = "grey"
color6 = "green"
color7 = "crimson"

dy = 1.5

ax[0].plot(t1*1e3, (g1       ), color=color1, linewidth=1.0, label="$g_1$")
ax[0].plot(t1*1e3, (g2 -   dy), color=color2, linewidth=1.0, label="$g_2$")
ax[0].plot(t1*1e3, (g3 - 2*dy), color=color3, linewidth=1.0, label="$g_3$")
ax[0].plot(t1*1e3, (g4 - 3*dy), color=color4, linewidth=1.0, label="$g_4$")
ax[0].plot(t1*1e3, (g5 - 4*dy), color=color5, linewidth=1.0, label="$g_5$")

ax[0].tick_params(left = False)
ax[0].set_yticks([])
