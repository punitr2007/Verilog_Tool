import numpy as np
import matplotlib.pyplot as plt 
import gseim_calc as calc
from setsize import set_size

f_hz = 50.0
T = 1.0/f_hz

slv = calc.slv("rectifier_1ph_12_orig.in")

i_slv = 0
i_out = 0
filename = slv.l_filename_all[i_slv][i_out]
print('filename:', filename)
u = np.loadtxt(filename)
t = u[:, 0]

v_sa = slv.get_array_double(i_slv,i_out,"v_sa",u)
v_sb = slv.get_array_double(i_slv,i_out,"v_sb",u)
v_o  = slv.get_array_double(i_slv,i_out,"v_o",u)

v_o_avg = calc.avg_3(t, v_o, 0.0, 2.0*T, 1.0e-5*T)

print('average value of v_o:', "%11.4E"%(v_o_avg))

l1 = calc.min_max_1(t, v_o, 0.0, 2.0*T)
v_o_ptop = l1[1] - l1[0]
print('ripple in v_o:', "%11.4E"%v_o_ptop)

color1 = 'blue'
color2 = 'green'
color3 = 'red'

fig, ax = plt.subplots(2, sharex=False)
plt.subplots_adjust(wspace=0, hspace=0.0)

set_size(5.5, 4, ax[0])

for i in range(2):
    ax[i].set_xlim(left=0.0, right=2.0*T*1e3)
    ax[i].grid(color='#CCCCCC', linestyle='solid', linewidth=0.5)

ax[0].set_ylabel(r'$v_s$'      , fontsize=12)
ax[1].set_ylabel(r'$v_o$'      , fontsize=12)

ax[0].tick_params(labelbottom=False)

ax[0].plot(t*1e3,  v_sa, color=color1, linewidth=1.0, label="$v_{sa}$")
ax[0].plot(t*1e3,  v_sb, color=color2, linewidth=1.0, label="$v_{sb}$")
ax[1].plot(t*1e3,  v_o , color=color3, linewidth=1.0, label="$v_o$")

ax[1].axhline(v_o_avg, color=color3, linewidth=1.0, label="$v_o^{avg}$", linestyle='--', dashes=(5,3))

ax[1].set_xlabel('time (msec)', fontsize=11)

for i in range(2):
    ax[i].legend(loc = 'lower right',frameon = True, fontsize = 10, title = None,
      markerfirst = True, markerscale = 1.0, labelspacing = 0.5, columnspacing = 2.0,
      prop = {'size' : 12},)

#plt.tight_layout()
plt.show()
