Push-pull converter

For the push-pull converter shown in the figure, the input voltage is $40\,$V, the output voltage is $20\,$V, and the load resistance is $2\,\Omega$. The gate signals $g_1$ and $g_2$ are shown in the figure. The magnetizing inductance as seen from the source side is $5\,$mH (for each winding). The output capacitance is $20\,\mu$F, and the filter inductance is $500\,\mu$H. Determine the following.
  1. the transformer turn ratio
  2. peak voltage stress on switches $S_1$, $S_2$, and the rms values of $i_{S1}$, $i_{S2}$
  3. peak voltage stress on diodes $D_1$, $D_2$, and the average values of $i_{D1}$, $i_{D2}$
  4. ripple frequency of the inductor current

import gseim_calc as calc

s_Vdc = "40"
s_L = "500e-6"
s_R = "2"
s_C = "20e-6"
s_n = "2.0" # to be changed by user

f_hz = 50.0e3
T = 1/f_hz
s_Tend = "3e-3"
Tend = float(s_Tend)
T1 = Tend - 2.0*T
s_T1 = ("%11.4E"%(T1)).strip()

l = [
  ('$Vdc', s_Vdc),
  ('$L', s_L),
  ('$R', s_R),
  ('$C', s_C),
  ('$n', s_n),
  ('$Tend', s_Tend),
  ('$T1', s_T1),
]
calc.replace_strings_1("push_pull_1_orig.in", "push_pull_1.in", l)
print('push_pull_1.in is ready for execution')

t = ta - ta[0]

v_out = slv.get_array_double(i_slv,i_out,"v_out",u)
IL    = slv.get_array_double(i_slv,i_out,"IL"   ,u)
VD1   = slv.get_array_double(i_slv,i_out,"VD1"  ,u)
VD2   = slv.get_array_double(i_slv,i_out,"VD2"  ,u)
VS1   = slv.get_array_double(i_slv,i_out,"VS1"  ,u)
VS2   = slv.get_array_double(i_slv,i_out,"VS2"  ,u)

ID1 = slv.get_array_double(i_slv,i_out,"ID1",u1)
ID2 = slv.get_array_double(i_slv,i_out,"ID2",u1)

IS2 = slv.get_array_double(i_slv,i_out,"IS2",u1)

# since we have stored two cycles, we need to divide the last time point
# by 2 to get the period:

T = t[-1]/2

max_VD1 = -min(VD1)
max_VD2 = -min(VD2)
max_VS1 =  max(VS1)
max_VS2 =  max(VS2)

print('max stress on D1:', "%11.4E"%(max_VD1), 'V')

print('max stress on S1:', "%11.4E"%(max_VS1), 'V')
print('max stress on S2:', "%11.4E"%(max_VS2), 'V')

l_ID1 = calc.avg_rms_3a(t1, ID1, 0.0, 2.0*T, 1.0e-3*T)
l_ID2 = calc.avg_rms_3a(t1, ID2, 0.0, 2.0*T, 1.0e-3*T)
l_IS1 = calc.avg_rms_3a(t1, IS1, 0.0, 2.0*T, 1.0e-3*T)
l_IS2 = calc.avg_rms_3a(t1, IS2, 0.0, 2.0*T, 1.0e-3*T)

print('average ID2:', "%11.4E"%l_ID2[0], 'A')

print('average IS2:', "%11.4E"%l_IS2[0], 'A')
print('rms IS1:'    , "%11.4E"%l_IS1[1], 'A')
print('rms IS2:'    , "%11.4E"%l_IS2[1], 'A')
