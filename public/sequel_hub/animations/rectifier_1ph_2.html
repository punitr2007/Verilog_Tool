import numpy as np
import matplotlib.pyplot as plt 
import gseim_calc as calc
from setsize import set_size

f_hz = 50.0
T = 1.0/f_hz

slv = calc.slv("rectifier_1ph_2.in")

i_slv = 0
i_out = 0
filename = slv.l_filename_all[i_slv][i_out]
print('filename:', filename)
u = np.loadtxt(filename)
t = u[:, 0]

ID1 = slv.get_array_double(i_slv,i_out,"ID1",u)
ID2 = slv.get_array_double(i_slv,i_out,"ID2",u)
vs  = slv.get_array_double(i_slv,i_out,"vs",u)

ID1_avg = calc.avg_3(t, ID1, 0.0, 2.0*T, 1.0e-5*T)
ID2_avg = calc.avg_3(t, ID2, 0.0, 2.0*T, 1.0e-5*T)

# IR and ID1 are the same:
print('average value of IR:', "%11.4E"%ID1_avg)

l_cross_1_ID1, l_cross_2_ID1 = calc.cross_over_points_1(t, ID1, 0.0, 2.0*T, 1.0e-10)
print('zero-crossing points of ID1 (positive slope):')
for t1 in l_cross_1_ID1:
    print("  ", "%11.4E"%t1)
print('zero-crossing points of ID1 (negative slope):')
for t1 in l_cross_2_ID1:
    print("  ", "%11.4E"%t1)
t_conduction_D1 = l_cross_2_ID1[1]-l_cross_1_ID1[0]
print("conduction duration for D1:", "%11.4E"%t_conduction_D1)

color1='blue'
color2='green'
color3='red'

fig, ax = plt.subplots(3, sharex=False)
plt.subplots_adjust(wspace=0, hspace=0.0)

set_size(5.5, 6, ax[0])

for i in range(3):
    ax[i].set_xlim(left=0.0, right=2.0*T*1e3)
    ax[i].grid(color='#CCCCCC', linestyle='solid', linewidth=0.5)

ax[0].set_ylabel(r'$v_s$',    fontsize=12)
ax[1].set_ylabel(r'$I_{D1}$', fontsize=12)
ax[2].set_ylabel(r'$I_{D2}$', fontsize=12)

ax[0].tick_params(labelbottom=False)
ax[1].tick_params(labelbottom=False)

ax[0].plot(t*1e3, vs , color=color1, linewidth=1.0, label="$v_s$")
ax[1].plot(t*1e3, ID1, color=color2, linewidth=1.0, label="$I_{D1}$")
ax[2].plot(t*1e3, ID2, color=color3, linewidth=1.0, label="$I_{D2}$")

ax[1].axhline(y=ID1_avg, color=color2, linewidth=1.0, label="$I_{D1}^{avg}$", linestyle='--', dashes=(5,3))
ax[2].axhline(y=ID2_avg, color=color3, linewidth=1.0, label="$I_{D2}^{avg}$", linestyle='--', dashes=(5,3))

ax[2].set_xlabel('time (msec)', fontsize=11)

for k in range(1, 3):
    ax[k].legend(loc = 'lower right',frameon = True, fontsize = 10, title = None,
      markerfirst = True, markerscale = 1.0, labelspacing = 0.5, columnspacing = 2.0,
      prop = {'size' : 12},)

#plt.tight_layout()
plt.show()
