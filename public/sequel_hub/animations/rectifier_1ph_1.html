import numpy as np
import matplotlib.pyplot as plt 
import gseim_calc as calc
from setsize import set_size

f_hz = 50.0
T = 1.0/f_hz

slv = calc.slv("rectifier_1ph_1_orig.in")

i_slv = 0
i_out = 0
filename = slv.l_filename_all[i_slv][i_out]
print('filename:', filename)
u = np.loadtxt(filename)
t = u[:, 0]

v_b = slv.get_array_double(i_slv,i_out,"v_b",u)
ID1 = slv.get_array_double(i_slv,i_out,"ID1",u)
ID2 = slv.get_array_double(i_slv,i_out,"ID2",u)

l_v_b = calc.avg_rms_3a(t, v_b, 0.0, 2.0*T, 1.0e-5*T)
l_ID1 = calc.avg_rms_3a(t, ID1, 0.0, 2.0*T, 1.0e-5*T)
l_ID2 = calc.avg_rms_3a(t, ID2, 0.0, 2.0*T, 1.0e-5*T)

print('average value of v_b:', "%11.4E"%l_v_b[0])
print('average value of ID1:', "%11.4E"%l_ID1[0])

color1='blue'
color2='olive'
color3='red'

fig, ax = plt.subplots(3, sharex=False)
plt.subplots_adjust(wspace=0, hspace=0.0)

set_size(5.5, 6, ax[0])

for i in range(3):
    ax[i].set_xlim(left=0.0, right=2.0*T*1e3)
    ax[i].grid(color='#CCCCCC', linestyle='solid', linewidth=0.5)

ax[0].set_ylabel(r'$V_b$',    fontsize=12)
ax[1].set_ylabel(r'$I_{D1}$', fontsize=12)
ax[2].set_ylabel(r'$I_{D2}$', fontsize=12)

ax[0].tick_params(labelbottom=False)
ax[1].tick_params(labelbottom=False)

ax[0].plot(t*1e3, v_b, color=color1, linewidth=1.0, label="$V_b$")
ax[0].axhline(l_v_b[0], color=color1, linewidth=1.0, label="$V_b^{avg}$", linestyle='--', dashes=(5,3))

ax[1].plot(t*1e3, ID1, color=color2, linewidth=1.0, label="$I_{D1}$")
ax[1].axhline(l_ID1[0], color=color2, linewidth=1.0, label="$I_{D1}^{avg}$", linestyle='--', dashes=(5,3))

ax[2].plot(t*1e3, ID2, color=color3, linewidth=1.0, label="$I_{D2}$")
ax[2].axhline(l_ID2[0], color=color3, linewidth=1.0, label="$I_{D2}^{avg}$", linestyle='--', dashes=(5,3))

ax[2].set_xlabel('time (msec)', fontsize=12)

for k in range(3):
    ax[k].legend(loc = 'lower right',frameon = True, fontsize = 10, title = None,
      markerfirst = True, markerscale = 1.0, labelspacing = 0.5, columnspacing = 2.0,
      prop = {'size' : 12},)

#plt.tight_layout()
plt.show()
